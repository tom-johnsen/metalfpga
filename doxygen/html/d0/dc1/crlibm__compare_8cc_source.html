64static uint64_t DoubleToBits(double value) {

70static double BitsToDouble(uint64_t bits) {

76static bool IsNaNBits(uint64_t bits) {
77 return ((bits >> 52) & 0x7FFu) == 0x7FFu &&
78 (bits & 0x000FFFFFFFFFFFFFull) != 0;
79}
80
81static bool IsInfBits(uint64_t bits) {
82 return ((bits >> 52) & 0x7FFu) == 0x7FFu &&
83 (bits & 0x000FFFFFFFFFFFFFull) == 0;
84}

86static uint64_t OrderedBits(uint64_t bits) {
87 const uint64_t sign = bits >> 63;
88 const uint64_t mask = sign ? 0xFFFFFFFFFFFFFFFFull : 0x8000000000000000ull;
89 return bits ^ mask;
90}

92static uint64_t ULPDiff(uint64_t a, uint64_t b) {
93 uint64_t oa = OrderedBits(a);
94 uint64_t ob = OrderedBits(b);
95 return (oa > ob) ? (oa - ob) : (ob - oa);
96}

98static bool DomainAccept(Domain domain, uint64_t bits) {
99 if (IsNaNBits(bits) || IsInfBits(bits)) {
100 return false;
101 }
102 double v = BitsToDouble(bits);
103 if (!std::isfinite(v)) {
104 return false;
105 }
106 switch (domain) {
107 case Domain::kAny:
108 return true;
109 case Domain::kPositive:
110 return v > 0.0;
111 case Domain::kNonNegative:
112 return v >= 0.0;
113 case Domain::kMinusOneToOne:
114 return v >= -1.0 && v <= 1.0;
115 case Domain::kGreaterMinusOne:
116 return v > -1.0;
117 }
118 return false;
119}

121static std::vector<uint64_t> EdgeInputs() {
122 return {

124 0x8000000000000000ull,

127 0x0010000000000000ull,
128 0x7FEFFFFFFFFFFFFFull,
129 0x7FF0000000000000ull,
130 0xFFF0000000000000ull,
131 0x7FF8000000000000ull,
132 0xFFF8000000000000ull,
133 DoubleToBits(1.0),
134 DoubleToBits(-1.0),
135 DoubleToBits(0.5),
136 DoubleToBits(-0.5),
137 DoubleToBits(2.0),
138 DoubleToBits(-2.0),
139 DoubleToBits(10.0),
140 DoubleToBits(-10.0),
141 DoubleToBits(3.14159265358979323846),
142 };

145static uint64_t RandomFinite(std::mt19937_64& rng) {
146 uint64_t sign = (rng() & 1ull) << 63;
147 uint64_t exp = rng() % 2047ull;
148 if (exp == 2047ull) {
149 exp = 2046ull;
150 }
151 uint64_t mant = rng() & 0x000FFFFFFFFFFFFFull;
152 return sign | (exp << 52) | mant;
153}

155static std::vector<uint64_t> MakeUnaryInputs(Domain domain, size_t count,
156 std::mt19937_64& rng) {
157 std::vector<uint64_t> inputs = EdgeInputs();
158 std::vector<uint64_t> filtered;
159 filtered.reserve(inputs.size());
160 for (uint64_t bits : inputs) {
161 if (DomainAccept(domain, bits) || IsNaNBits(bits) || IsInfBits(bits)) {
162 filtered.push_back(bits);
163 }
164 }
165 while (filtered.size() < count) {
166 uint64_t bits = RandomFinite(rng);
167 if (DomainAccept(domain, bits)) {
168 filtered.push_back(bits);
169 }
170 }
171 return filtered;
172}

174static std::vector<std::pair<uint64_t, uint64_t>> MakePowInputs(size_t count,
175 std::mt19937_64& rng) {
176 std::vector<uint64_t> edges = EdgeInputs();
177 std::vector<std::pair<uint64_t, uint64_t>> inputs;
178 for (uint64_t a : edges) {
179 inputs.emplace_back(a, DoubleToBits(2.0));
180 inputs.emplace_back(a, DoubleToBits(-2.0));
181 inputs.emplace_back(a, DoubleToBits(0.5));
182 inputs.emplace_back(a, DoubleToBits(-0.5));
183 inputs.emplace_back(a, DoubleToBits(0.0));
184 inputs.emplace_back(a, DoubleToBits(1.0));
185 }
186 while (inputs.size() < count) {
187 uint64_t base = RandomFinite(rng);
188 uint64_t exp = RandomFinite(rng);
189 inputs.emplace_back(base, exp);
190 }
191 return inputs;
192}

331 uint64_t seed = 1;

333 std::string mode_arg = "rn";
334 std::string out_dir;
335 bool trace = false;
336
337 for (int i = 1; i < argc; ++i) {
338 std::string arg = argv[i];
339 if (arg == "--count" && i + 1 < argc) {
340 count = static_cast<size_t>(std::stoull(argv[++i]));
341 } else if (arg == "--seed" && i + 1 < argc) {
342 seed = static_cast<uint64_t>(std::stoull(argv[++i]));
343 } else if (arg == "--func" && i + 1 < argc) {
344 func_arg = argv[++i];
345 } else if (arg == "--mode" && i + 1 < argc) {
346 mode_arg = argv[++i];
347 } else if (arg == "--out-dir" && i + 1 < argc) {
348 out_dir = argv[++i];
349 } else if (arg == "--trace") {
350 trace = true;
351 } else if (arg == "--help") {
352 std::cout << "Usage: metalfpga_crlibm_compare [--func list|all] "
353 "[--mode rn|rd|ru|rz|all] [--count N] [--seed N] "
354 "[--out-dir path] [--trace]\n";
355 return 0;
356 }
357 }
358
359 if (out_dir.empty()) {
360 std::random_device rd;
361 uint64_t tag_seed =
362 (static_cast<uint64_t>(rd()) << 32) ^ static_cast<uint64_t>(rd());
363 if (tag_seed == 0) {
364 tag_seed = static_cast<uint64_t>(
365 std::chrono::high_resolution_clock::now().time_since_epoch().count());
366 }
367 uint32_t tag = static_cast<uint32_t>(tag_seed ^ (tag_seed >> 32));
368 std::ostringstream oss;

370 << tag;

374 std::mt19937_64 rng(seed);
375 fs::path out_path(out_dir);
376 if (!EnsureDir(out_path)) {

381 if (trace) {

383 }
384

388 if (!csv) {
389 std::cerr << "failed to open results.csv\n";
390 return 1;
391 }
392 csv << "func,mode,input0,input1,ref,got,ulp,status\n";

395 std::vector<UnarySpec> unary_specs = BuildUnarySpecs();
396 std::vector<BinarySpec> binary_specs = BuildBinarySpecs();
397
398 std::vector<Mode> modes;
399 if (mode_arg == "all") {
400 modes = {Mode::kRn, Mode::kRd, Mode::kRu, Mode::kRz};
401 } else {
402 Mode m;
403 if (!ParseMode(mode_arg, &m)) {
404 std::cerr << "unknown mode: " << mode_arg << "\n";
405 return 1;
406 }
407 modes = {m};
408 }
409
410 std::unordered_map<std::string, CompareStats> stats;
411
412 auto want_func = [&](const std::string& name) -> bool {
413 if (func_arg == "all") {
414 return true;
415 }
416 std::istringstream iss(func_arg);
417 std::string token;
418 while (std::getline(iss, token, ',')) {
419 if (token == name) {
420 return true;
421 }
422 }
423 return false;

425
426 const unsigned long long fpu_token = crlibm_init();
427
428 for (const auto& spec : unary_specs) {
429 if (!want_func(spec.name)) {
430 continue;
431 }
432 for (Mode mode : modes) {

434 auto gpga = SelectGpga(spec, mode);
435 if (!ref || !gpga) {
436 continue;
437 }
438 std::vector<uint64_t> inputs = MakeUnaryInputs(spec.domain, count, rng);
439 CompareStats stat;
440 for (uint64_t in_bits : inputs) {
441 double ref_val = ref(BitsToDouble(in_bits));
442 uint64_t ref_bits = DoubleToBits(ref_val);
443 gpga_double gpga_bits = gpga(static_cast<gpga_double>(in_bits));
444 uint64_t got_bits = static_cast<uint64_t>(gpga_bits);
445
446 uint64_t ulp = 0;
447 bool ok = false;
448 std::string status = "ok";
449 if (IsNaNBits(ref_bits) && IsNaNBits(got_bits)) {
450 ok = true;
451 } else if (IsInfBits(ref_bits) || IsInfBits(got_bits)) {
452 ok = (ref_bits == got_bits);

508 bool ok = false;
509 std::string status = "ok";
510 if (IsNaNBits(ref_bits) && IsNaNBits(got_bits)) {
511 ok = true;
512 } else if (IsInfBits(ref_bits) || IsInfBits(got_bits)) {
513 ok = (ref_bits == got_bits);
514 if (!ok) {
515 status = "inf_mismatch";
516 ulp = std::numeric_limits<uint64_t>::max();
517 }
518 } else {
519 ulp = ULPDiff(ref_bits, got_bits);
520 ok = (ulp == 0);
521 if (!ok) {
522 status = "ulp_mismatch";
523 }
524 }
525 stat.total += 1;
526 if (ok) {
527 stat.pass += 1;
528 } else {
529 stat.fail += 1;
530 if (ulp > stat.max_ulp) {
531 stat.max_ulp = ulp;
532 stat.worst_input0 = a_bits;
533 stat.worst_input1 = b_bits;
534 stat.worst_ref = ref_bits;
535 stat.worst_got = got_bits;
536 }
537 }
538 stat.sum_ulp += static_cast<long double>(ulp);
539 if (trace) {
540 csv << spec.name << ",rn," << HexU64(a_bits) << "," << HexU64(b_bits)
541 << "," << HexU64(ref_bits) << "," << HexU64(got_bits) << "," << ulp
542 << "," << status << "\n";
543 }
544 }
545 stats[spec.name + ":rn"] = stat;
546 }
547
548 crlibm_exit(fpu_token);

551 summary << "{\n";
552 summary << " \"seed\": " << seed << ",\n";
553 summary << " \"count\": " << count << ",\n";
554 summary << " \"results\": [\n";

558 summary << ",\n";
