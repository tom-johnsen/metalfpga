81inline std::string MslMangleIdentifier(std::string_view name) {
82 bool needs_escape = name.empty();
83 if (!needs_escape) {
84 if (!IsMslIdentStart(name.front())) {
85 needs_escape = true;
86 } else {
87 for (char c : name) {
88 if (!IsMslIdentChar(c)) {
89 needs_escape = true;
90 break;
91 }
92 }
93 }
94 }
95 if (!needs_escape) {
96 if (IsMslReservedIdentifier(name) || StartsWith(name, "__gpga_") ||
97 StartsWith(name, "gpga_")) {
98 needs_escape = true;
99 }
100 }
101 if (!needs_escape) {
102 return std::string(name);
103 }
104 std::string sanitized;
105 sanitized.reserve(name.size());
106 for (char c : name) {
107 sanitized.push_back(IsMslIdentChar(c) ? c : '_');
108 }
109 if (sanitized.empty()) {
110 sanitized = "id";
111 }
112 if (!IsMslIdentStart(sanitized.front())) {
113 sanitized.insert(sanitized.begin(), '_');
114 }
115 uint64_t hash = Fnv1aHash64(name);
116 return "__gpga_u_" + sanitized + "_" + Hex64(hash);
117}
