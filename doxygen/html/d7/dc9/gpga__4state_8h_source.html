144inline FourState32 fs_sext32(FourState32 a, uint src_width, uint target_width) {
145 if (target_width == 0u || src_width == 0u) return fs_make32(0u, 0u, target_width);
146 if (target_width <= src_width) return fs_make32(a.val, a.xz, target_width);
