544inline bool BuildSchedulerVmLayout(
545 const std::vector<std::vector<uint32_t>>& procs,
546 SchedulerVmLayout* out, std::string* error) {
547 if (!out) {
548 if (error) {
549 *error = "missing scheduler VM layout output";
550 }
551 return false;
552 }
553 out->proc_count = 0u;
554 out->words_per_proc = 0u;
555 out->bytecode.clear();
556 out->proc_offsets.clear();
557 out->proc_lengths.clear();
558 out->packed_slots.clear();
559 out->signal_entries.clear();
560 out->cond_entries.clear();
561 out->case_headers.clear();
562 out->case_entries.clear();
563 out->case_words.clear();
564 out->assign_entries.clear();
565 out->delay_assign_entries.clear();
566 out->force_entries.clear();
567 out->release_entries.clear();
568 out->service_entries.clear();
569 out->service_args.clear();
570 out->service_ret_entries.clear();
571 out->expr_table.words.clear();
572 out->expr_table.imm_words.clear();
573 out->edge_item_expr_offsets.clear();
574 out->edge_star_expr_offsets.clear();
575 const uint32_t proc_count = static_cast<uint32_t>(procs.size());
576 if (proc_count == 0u) {
577 out->proc_count = 0u;
578 out->words_per_proc = kSchedulerVmWordsPerProc;
579 return true;
580 }
581 uint32_t max_len = 0u;
582 for (const auto& proc : procs) {
583 max_len = std::max<uint32_t>(max_len,
584 static_cast<uint32_t>(proc.size()));
585 }
586 const uint32_t words_per_proc =
587 std::max<uint32_t>(max_len, kSchedulerVmWordsPerProc);
588 out->proc_count = proc_count;
589 out->words_per_proc = words_per_proc;
590 out->bytecode.assign(static_cast<size_t>(proc_count) * words_per_proc, 0u);
591 out->proc_offsets.resize(proc_count);
592 out->proc_lengths.resize(proc_count);
593 for (uint32_t pid = 0u; pid < proc_count; ++pid) {
594 const uint32_t offset = pid * words_per_proc;
595 out->proc_offsets[pid] = offset;
596 out->proc_lengths[pid] =
597 static_cast<uint32_t>(procs[pid].size());
598 if (procs[pid].empty()) {
599 continue;
600 }
601 std::copy(procs[pid].begin(), procs[pid].end(),
602 out->bytecode.begin() + offset);
603 }
604 return true;
605}

615inline bool BuildSchedulerVmSeedLayout(uint32_t proc_count,
616 SchedulerVmLayout* out,
617 std::string* error) {
618 std::vector<std::vector<uint32_t>> procs(proc_count);
619 for (uint32_t pid = 0u; pid < proc_count; ++pid) {
620 SchedulerVmBuilder builder;
621 builder.EmitCallGroup();
622 builder.EmitDone();
623 procs[pid] = builder.words();
624 }
625 return BuildSchedulerVmLayout(procs, out, error);
626}
