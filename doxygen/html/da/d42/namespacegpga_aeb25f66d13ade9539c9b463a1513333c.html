1007 }

1009 }

1011 out.width = 0;
1012 out.value_bits = 0;
1013 out.x_bits = 0;
1014 out.z_bits = 0;
1015 auto append = [&](const FourStateValue& part) {
1016 int part_width = ValueWidth(part);
1017 if (part_width <= 0) {
1018 return;
1019 }
1020 if (part_width > 64) {
1021 part_width = 64;
1022 }
1023 if (out.width >= 64) {
1024 out.width = 64;
1025 return;
1026 }
1027 uint64_t part_mask = MaskForWidth(part_width);
1028 uint64_t part_val = part.value_bits & part_mask;
1029 uint64_t part_x = part.x_bits & part_mask;
1030 uint64_t part_z = part.z_bits & part_mask;

1032 out.value_bits = (out.value_bits << part_width) | part_val;
1033 out.x_bits = (out.x_bits << part_width) | part_x;
1034 out.z_bits = (out.z_bits << part_width) | part_z;
1035 out.width += part_width;
1036 return;
1037 }
1038 int keep = 64 - out.width;
1039 if (keep <= 0) {
1040 out.width = 64;
1041 return;
1042 }
1043 uint64_t keep_mask = MaskForWidth(keep);
1044 out.value_bits = (out.value_bits << keep) | (part_val & keep_mask);
1045 out.x_bits = (out.x_bits << keep) | (part_x & keep_mask);
1046 out.z_bits = (out.z_bits << keep) | (part_z & keep_mask);
1047 out.width = 64;
1048 };

1050 for (const auto& element : expr.elements) {
1051 if (!element) {
1052 continue;
1053 }
1054 FourStateValue element_value;
1055 if (!EvalConstExpr4State(*element, params, &element_value, error)) {
1056 return false;
1057 }
1058 append(element_value);
1059 }
1060 }
1061 if (out.width == 0) {
1062 out.width = 1;
1063 }
1064 *out_value = out;
1065 return true;
1066 }

1068 if (error) {
1069 *error = "function call not allowed in constant expression";
1070 }
1071 return false;
1072 }
